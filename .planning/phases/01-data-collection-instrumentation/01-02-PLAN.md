class LanggraphCollector:
    """Collects LangGraph execution events via astream_events.

    Non-invasive event capture that doesn't modify agent code.
    Uses async streaming to avoid blocking the trading pipeline.
    """

    def __init__(self, run_id: Optional[str] = None):
        self.run_id = run_id or str(uuid.uuid4())
        self.events: List[AgentEvent] = []

    async def collect_events(
        self,
        graph: StateGraph,
        input_state: Dict[str, Any],
        config: Optional[Dict[str, Any]] = None
    ) -> AsyncIterator[AgentEvent]:
        """Stream events from graph execution.

        Yields AgentEvent objects as they occur.
        Non-blocking async implementation.
        """
        config = config or {}
        config.setdefault("run_name", f"observability_{self.run_id}")

        async for event in graph.astream_events(
            input_state,
            version="v1",  # Required for astream_events
            config=config
        ):
            # Filter relevant events (don't capture everything)
            if event["event"] in [
                "on_chat_model_start",
                "on_chat_model_end",
                "on_tool_start",
                "on_tool_end",
                "on_chain_start",
                "on_chain_end"
            ]:
                agent_event = self._parse_langgraph_event(event)
                if agent_event:
                    self.events.append(agent_event)
                    yield agent_event
```

Implement _parse_langgraph_event() to convert LangGraph events to AgentEvent models:

Key requirements:

class StateExtractor:
    """Extracts decision records from AgentState.

    Converts the nested AgentState structure into DecisionRecord models
    with proper signal extraction and debate capture.
    """

    def __init__(self):
        self.signal_extractor = TradingSignal.from_string

    def extract_decision_records(
        self,
        final_state: AgentState,
        ticker: str,
        trade_date: str,
        run_id: Optional[str] = None
    ) -> List[DecisionRecord]:
        """Extract all decision records from final AgentState.

        Returns a list of DecisionRecord objects, one per agent
        that made a decision during execution.
        """
        records = []

        # Extract analyst decisions
        records.extend(self._extract_analyst_decisions(final_state, ticker, trade_date, run_id))

        # Extract researcher decisions
        records.extend(self._extract_researcher_decisions(final_state, ticker, trade_date, run_id))

        # Extract trader decision
        records.append(self._extract_trader_decision(final_state, ticker, trade_date, run_id))

        # Extract risk judge decision
        records.append(self._extract_risk_decision(final_state, ticker, trade_date, run_id))

        # Extract final decision
        records.append(self._extract_final_decision(final_state, ticker, trade_date, run_id))

        return records
```
