<!-- From 02-RESEARCH.md - calibration pattern -->
```python
from sklearn.calibration import calibration_curve
from sklearn.metrics import brier_score_loss

class CalibrationTracker:
    def __init__(self):
        self.confidences: List[float] = []
        self.outcomes: List[bool] = []

    def expected_calibration_error(self, n_bins: int = 10) -> float:
        prob_true, prob_pred = calibration_curve(self.outcomes, self.confidences, n_bins=n_bins)
        ece = 0.0
        for i in range(len(prob_true)):
            ece += abs(prob_true[i] - prob_pred[i]) / len(prob_true)
        return ece
```
